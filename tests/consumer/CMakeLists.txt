cmake_minimum_required(VERSION 3.28)
if(ORBITFIT_PARENT_BUILD)
  load_cache("${ORBITFIT_PARENT_BUILD}" READ_WITH_PREFIX parent_ CMAKE_CXX_COMPILER CMAKE_PREFIX_PATH)
  set(CMAKE_CXX_COMPILER "${parent_CMAKE_CXX_COMPILER}")
  list(APPEND CMAKE_PREFIX_PATH "${parent_CMAKE_PREFIX_PATH}")
endif()
project(orbitfit_consumer LANGUAGES CXX)
find_package(orbitfit 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE orbitfit::orbitfit)
enable_testing()
add_test(NAME installed_consumer COMMAND consumer)
